analisar_variavel <- function(dados, variavel) {
  
  
  cat("\n\n")
  cat("====================================================\n")
  cat("ANÁLISE:", variavel, "\n")
  cat("====================================================\n")
  
  
  # -------------------------------------------------------
  # 3.1 ESTATÍSTICA DESCRITIVA + IC95%
  # -------------------------------------------------------
  
  desc <- dados %>%
    group_by(Grupo) %>%
    summarise(
      n = sum(!is.na(.data[[variavel]])),
      media = mean(.data[[variavel]], na.rm = TRUE),
      mediana = median(.data[[variavel]], na.rm = TRUE),
      dp = sd(.data[[variavel]], na.rm = TRUE),
      minimo = min(.data[[variavel]], na.rm = TRUE),
      maximo = max(.data[[variavel]], na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      erro_padrao = dp / sqrt(n),
      t_critico = qt(0.975, df = n - 1),
      IC_inf = media - t_critico * erro_padrao,
      IC_sup = media + t_critico * erro_padrao
    )
  
  
  cat("\n--- Estatística descritiva ---\n")
  print(desc)
  
  
  # -------------------------------------------------------
  # 3.2 MODELO ANOVA
  # -------------------------------------------------------
  
  formula <- as.formula(
    paste(variavel, "~ Grupo")
  )
  
  modelo <- aov(formula, data = dados)
  
  
  # -------------------------------------------------------
  # 3.3 NORMALIDADE DOS RESÍDUOS
  # -------------------------------------------------------
  
  shapiro <- shapiro.test(residuals(modelo))
  p_shapiro <- shapiro$p.value
  
  cat("\n--- Pressuposto de normalidade dos resíduos ---\n")
  cat(
    "Shapiro-Wilk: p =",
    format.pval(p_shapiro),
    "\n"
  )
  
  if (p_shapiro < 0.05) {
    
    cat(
      "Interpretação: Os dados diferem significativamente de uma distribuição normal. Rejeita-se a normalidade.\n",
      sep = ""
    )
    
  } else {
    
    cat(
      "Interpretação: Os dados podem ser considerados normais\n"
    )
  }
  
  
  # -------------------------------------------------------
  # 3.4 HOMOGENEIDADE DAS VARIÂNCIAS
  # -------------------------------------------------------
  
  levene <- leveneTest(formula, data = dados)
  p_levene <- levene$`Pr(>F)`[1]
  
  cat("\n--- Pressuposto de homogeneidade das variâncias ---\n")
  cat(
    "Levene: p =",
    format.pval(p_levene),
    "\n"
  )
  
  if (p_levene < 0.05) {
    
    cat(
      "Interpretação: há evidências de que as variâncias ",
      "dos grupos são diferentes.\n",
      sep = ""
    )
    
  } else {
    
    cat(
      "Interpretação: não há evidências suficientes de ",
      "diferença entre as variâncias.\n"
    )
  }
  
  
  # -------------------------------------------------------
  # 3.5 ANOVA
  # -------------------------------------------------------
  
  anova_resultado <- summary(modelo)
  
  p_anova <- anova_resultado[[1]]$`Pr(>F)`[1]
  
  cat("\n--- ANOVA ---\n")
  print(anova_resultado)
  
  if (p_anova < 0.05) {
    
    cat(
      "\nInterpretação: há evidências de diferença ",
      "entre as médias dos grupos.\n",
      sep = ""
    )
    
  } else {
    
    cat(
      "\nInterpretação: não há evidências suficientes ",
      "de diferença entre as médias dos grupos.\n"
    )
  }
  
  
  # -------------------------------------------------------
  # 3.6 TAMANHO DE EFEITO - ETA²
  # -------------------------------------------------------
  
  tabela_anova <- anova_resultado[[1]]
  
  SQ_grupo <- tabela_anova["Grupo", "Sum Sq"]
  SQ_total <- sum(tabela_anova[, "Sum Sq"])
  
  eta2 <- SQ_grupo / SQ_total
  
  cat("\n--- Tamanho de efeito ---\n")
  cat("Eta² =", round(eta2, 4), "\n")
  
  
  # -------------------------------------------------------
  # 3.7 TUKEY
  # -------------------------------------------------------
  
  tukey <- NULL
  
  if (p_anova < 0.05) {
    
    cat("\n--- Comparações múltiplas de Tukey ---\n")
    
    tukey <- TukeyHSD(modelo)
    
    print(tukey)
    
    tukey_df <- as.data.frame(tukey$Grupo)
    tukey_df$comparacao <- rownames(tukey_df)
    
    significativos <- tukey_df %>%
      filter(`p adj` < 0.05)
    
    cat("\nComparações significativas:\n")
    
    if (nrow(significativos) == 0) {
      
      cat("Nenhum par apresentou diferença significativa.\n")
      
    } else {
      
      for (i in 1:nrow(significativos)) {
        
        cat(
          significativos$comparacao[i],
          "→ p =",
          format.pval(significativos$`p adj`[i]),
          "\n"
        )
      }
    }
    
  } else {
    
    cat(
      "\nTukey não realizado, pois a ANOVA ",
      "não foi significativa.\n",
      sep = ""
    )
  }
  
  
  # -------------------------------------------------------
  # 3.8 CONTRASTE PLANEJADO
  #     E VS MÉDIA DOS GRUPOS A, B, C E D
  # -------------------------------------------------------
  
  cat(
    "\n--- Contraste planejado: E vs. média de A-D ---\n"
  )
  
  medias <- emmeans(modelo, ~ Grupo)
  
  contraste_E <- contrast(
    medias,
    method = list(
      "E vs. A-D" = c(
        -0.25,
        -0.25,
        -0.25,
        -0.25,
        1
      )
    )
  )
  
  contraste_resultado <- summary(
    contraste_E,
    infer = c(TRUE, TRUE)
  )
  
  print(contraste_resultado)
  
  
  # Valores do contraste
  
  estimativa <- contraste_resultado$estimate
  p_contraste <- contraste_resultado$p.value
  IC_inf_contraste <- contraste_resultado$lower.CL
  IC_sup_contraste <- contraste_resultado$upper.CL
  
  
  # -------------------------------------------------------
  # 3.9 INTERPRETAÇÃO DO CONTRASTE
  # -------------------------------------------------------
  
  if (p_contraste < 0.05) {
    
    if (estimativa > 0) {
      
      cat(
        "\nInterpretação: o grupo E apresenta média ",
        "significativamente maior que a média dos grupos ",
        "A, B, C e D.\n",
        sep = ""
      )
      
    } else {
      
      cat(
        "\nInterpretação: o grupo E apresenta média ",
        "significativamente menor que a média dos grupos ",
        "A, B, C e D.\n",
        sep = ""
      )
    }
    
  } else {
    
    cat(
      "\nInterpretação: não há evidências suficientes de ",
      "diferença entre a média do grupo E e a média dos ",
      "grupos A, B, C e D.\n",
      sep = ""
    )
  }
  
  
  # -------------------------------------------------------
  # 3.10 TAMANHO DE EFEITO DO CONTRASTE
  # -------------------------------------------------------
  
  sigma_modelo <- sigma(modelo)
  
  d_contraste <- estimativa / sigma_modelo
  
  cat("\n--- Tamanho de efeito do contraste E vs. A-D ---\n")
  cat(
    "d =",
    round(d_contraste, 4),
    "\n"
  )
  
  
  # -------------------------------------------------------
  # 3.11 OBSERVAÇÃO SOBRE OS PRESSUPOSTOS
  # -------------------------------------------------------
  
  if (p_shapiro < 0.05 | p_levene < 0.05) {
    
    cat(
      "\n--- Observação ---\n"
    )
    
    cat(
      "Um ou mais pressupostos da ANOVA apresentaram ",
      "evidências de violação. Os resultados da ANOVA ",
      "devem, portanto, ser interpretados com cautela.\n",
      sep = ""
    )
  }
  
  
  # -------------------------------------------------------
  # 3.12 GRÁFICOS DE DIAGNÓSTICO
  # -------------------------------------------------------
  
  par(mfrow = c(2, 2))
  plot(modelo)
  par(mfrow = c(1, 1))
  
  
  # -------------------------------------------------------
  # 3.13 RETORNO DOS RESULTADOS
  # -------------------------------------------------------
  
  return(
    list(
      descritiva = desc,
      modelo = modelo,
      shapiro = shapiro,
      levene = levene,
      anova = anova_resultado,
      tukey = tukey,
      contraste_E = contraste_resultado,
      eta2 = eta2,
      d_contraste = d_contraste
    )
  )
}

grafico_media_ic <- function(dados, variavel) {
  
  resumo <- dados %>%
    group_by(Grupo) %>%
    summarise(
      n = sum(!is.na(.data[[variavel]])),
      media = mean(.data[[variavel]], na.rm = TRUE),
      dp = sd(.data[[variavel]], na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      erro = qt(0.975, df = n - 1) * dp / sqrt(n),
      IC_inf = media - erro,
      IC_sup = media + erro
    )
  
  
  ggplot(
    resumo,
    aes(x = Grupo, y = media)
  ) +
    geom_point(size = 3) +
    geom_errorbar(
      aes(
        ymin = IC_inf,
        ymax = IC_sup
      ),
      width = 0.15
    ) +
    labs(
      title = paste(
        "Média de", variavel,
        "por classe de umidade"
      ),
      x = "Classe de umidade",

analisar_sem_outliers <- function(dados, variavel) {
  
  
  cat("\n\n")
  cat("==========================================\n")
  cat("VARIÁVEL:", variavel, "\n")
  cat("==========================================\n")
  
  
  # =======================================================
  # ESTATÍSTICAS DESCRITIVAS
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("1. ESTATÍSTICAS DESCRITIVAS\n")
  cat("------------------------------------------\n")
  
  medias <- dados %>%
    group_by(Grupo) %>%
    summarise(
      n = sum(!is.na(.data[[variavel]])),
      media = mean(.data[[variavel]], na.rm = TRUE),
      mediana = median(.data[[variavel]], na.rm = TRUE),
      dp = sd(.data[[variavel]], na.rm = TRUE),
      minimo = min(.data[[variavel]], na.rm = TRUE),
      maximo = max(.data[[variavel]], na.rm = TRUE),
      .groups = "drop"
    )
  
  print(medias)
  
  
  cat("\nINTERPRETAÇÃO:\n")
  cat(
    "As estatísticas descritivas mostram como os valores de",
    variavel,
    "se distribuem dentro de cada grupo.\n"
  )
  
  cat(
    "A média permite comparar o valor central das condições,\n",
    "enquanto o desvio-padrão mostra a dispersão dos valores\n",
    "dentro de cada grupo.\n"
  )
  
  
  # =======================================================
  # INTERVALOS DE CONFIANÇA
  # =======================================================
  
  medias <- medias %>%
    mutate(
      erro_padrao = dp / sqrt(n),
      t_critico = qt(0.975, df = n - 1),
      IC95_inf = media - t_critico * erro_padrao,
      IC95_sup = media + t_critico * erro_padrao
    )
  
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("2. INTERVALOS DE CONFIANÇA DE 95%\n")
  cat("------------------------------------------\n")
  
  print(
    medias %>%
      select(Grupo, n, media, IC95_inf, IC95_sup)
  )
  
  
  cat("\nINTERPRETAÇÃO:\n")
  cat(
    "O intervalo de confiança fornece uma faixa de valores\n",
    "plausíveis para a média de cada grupo, considerando 95%\n",
    "de confiança.\n"
  )
  
  cat(
    "Intervalos mais estreitos indicam maior precisão na\n",
    "estimativa da média, enquanto intervalos mais largos\n",
    "indicam maior incerteza.\n"
  )
  
  
  # =======================================================
  # MODELO ANOVA
  # =======================================================
  
  formula <- as.formula(
    paste(variavel, "~ Grupo")
  )
  
  modelo <- aov(
    formula,
    data = dados
  )
  
  
  # =======================================================
  # ANOVA
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("3. ANOVA\n")
  cat("------------------------------------------\n")
  
  tabela_anova <- summary(modelo)[[1]]
  
  print(tabela_anova)
  
  
  p_anova <- tabela_anova[
    "Grupo",
    "Pr(>F)"
  ]
  
  
  cat("\nINTERPRETAÇÃO:\n")
  
  if (p_anova < 0.05) {
    
    cat(
      "O p-valor da ANOVA é",
      signif(p_anova, 4),
      "e é menor que 0,05.\n"
    )
    
    cat(
      "Portanto, há evidência estatística de que pelo menos\n",
      "uma das médias dos grupos seja diferente das demais.\n"
    )
    
    cat(
      "A ANOVA, sozinha, não informa quais grupos diferem.\n",
      "Por isso, quando significativa, realizamos o teste de Tukey.\n"
    )
    
  } else {
    
    cat(
      "O p-valor da ANOVA é",
      signif(p_anova, 4),
      "e é maior ou igual a 0,05.\n"
    )
    
    cat(
      "Portanto, não há evidência estatística suficiente para\n",
      "afirmar que as médias dos grupos sejam diferentes.\n"
    )
  }
  
  
  # =======================================================
  # SHAPIRO-WILK
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("4. TESTE DE NORMALIDADE - SHAPIRO-WILK\n")
  cat("------------------------------------------\n")
  
  shapiro <- shapiro.test(
    residuals(modelo)
  )
  
  print(shapiro)
  
  
  cat("\nINTERPRETAÇÃO:\n")
  
  if (shapiro$p.value < 0.05) {
    
    cat(
      "O p-valor é",
      signif(shapiro$p.value, 4),
      "e é menor que 0,05.\n"
    )
    
    cat(
      "Portanto, há evidência de que os resíduos não seguem\n",
      "uma distribuição normal.\n"
    )
    
    cat(
      "Esse resultado representa uma violação do pressuposto\n",
      "de normalidade da ANOVA.\n"
    )
    
  } else {
    
    cat(
      "O p-valor é",
      signif(shapiro$p.value, 4),
      "e é maior que 0,05.\n"
    )
    
    cat(
      "Não há evidência estatística de violação da normalidade\n",
      "dos resíduos.\n"
    )
  }
  
  
  # =======================================================
  # LEVENE
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("5. HOMOGENEIDADE DAS VARIÂNCIAS - LEVENE\n")
  cat("------------------------------------------\n")
  
  levene <- leveneTest(
    formula,
    data = dados
  )
  
  print(levene)
  
  p_levene <- levene$`Pr(>F)`[1]
  
  
  cat("\nINTERPRETAÇÃO:\n")
  
  if (p_levene < 0.05) {
    
    cat(
      "O p-valor é",
      signif(p_levene, 4),
      "e é menor que 0,05.\n"
    )
    
    cat(
      "Há evidência de que as variâncias dos grupos são diferentes.\n"
    )
    
    cat(
      "Portanto, o pressuposto de homogeneidade das variâncias\n",
      "não foi atendido.\n"
    )
    
  } else {
    
    cat(
      "O p-valor é",
      signif(p_levene, 4),
      "e é maior que 0,05.\n"
    )
    
    cat(
      "Não há evidência estatística de que as variâncias dos\n",
      "grupos sejam diferentes.\n"
    )
    
    cat(
      "Portanto, o pressuposto de homogeneidade das variâncias\n",
      "é considerado atendido.\n"
    )
  }
  
  
  # =======================================================
  # TAMANHO DE EFEITO - ETA²
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("6. TAMANHO DE EFEITO - ETA²\n")
  cat("------------------------------------------\n")
  
  SQ_grupo <- tabela_anova[
    "Grupo",
    "Sum Sq"
  ]
  
  SQ_total <- sum(
    tabela_anova[, "Sum Sq"]
  )
  
  eta2 <- SQ_grupo / SQ_total
  
  cat(
    "eta² =",
    round(eta2, 4),
    "\n"
  )
  
  
  cat("\nINTERPRETAÇÃO:\n")
  
  cat(
    "O eta² representa a proporção da variabilidade total\n",
    "da variável que está associada às diferenças entre os grupos.\n"
  )
  
  cat(
    "Por exemplo, um eta² de 0,50 significa que aproximadamente\n",
    "50% da variabilidade observada está associada às diferenças\n",
    "entre os grupos dentro desse modelo.\n"
  )
  
  cat(
    "O eta² deve ser interpretado junto com o p-valor da ANOVA\n",
    "e com os pressupostos do modelo.\n"
  )
  
  
  # =======================================================
  # TUKEY
  # =======================================================
  
  if (p_anova < 0.05) {
    
    cat("\n")
    cat("------------------------------------------\n")
    cat("7. TESTE DE TUKEY\n")
    cat("------------------------------------------\n")
    
    tukey <- TukeyHSD(modelo)
    
    print(tukey)
    
    
    cat("\nINTERPRETAÇÃO:\n")
    
    cat(
      "Como a ANOVA foi significativa, o teste de Tukey foi\n",
      "utilizado para identificar quais pares de grupos apresentam\n",
      "diferenças estatisticamente significativas.\n"
    )
    
    cat(
      "Nas comparações, valores de p ajustado menores que 0,05\n",
      "indicam diferença significativa entre os respectivos grupos.\n"
    )
    
  } else {
    
    tukey <- NULL
    
    cat("\n")
    cat("------------------------------------------\n")
    cat("7. TESTE DE TUKEY\n")
    cat("------------------------------------------\n")
    
    cat(
      "O Tukey não foi realizado porque a ANOVA não apresentou\n",
      "diferença estatisticamente significativa.\n"
    )
  }
  
  
  # =======================================================
  # CONTRASTE PLANEJADO E VS A-D
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("8. CONTRASTE PLANEJADO: E VS A-D\n")
  cat("------------------------------------------\n")
  
  medias_emmeans <- emmeans(
    modelo,
    ~ Grupo
  )
  
  
  contraste_E <- contrast(
    medias_emmeans,
    method = list(
      "E vs. A-D" =
        c(
          1,
          -0.25,
          -0.25,
          -0.25,
          -0.25
        )
    )
  )
  
  
  resultado_contraste <- summary(
    contraste_E,
    infer = c(TRUE, TRUE)
  )
  
  print(resultado_contraste)
  
  
  p_contraste <- resultado_contraste$p.value
  estimativa <- resultado_contraste$estimate
  
  
  cat("\nINTERPRETAÇÃO:\n")
  
  cat(
    "Esse contraste compara a média do grupo E com a média\n",
    "dos grupos A, B, C e D.\n"
  )
  
  cat(
    "A diferença calculada é:\n",
    "E - média(A,B,C,D)\n"
  )
  
  
  if (estimativa > 0) {
    
    cat(
      "Como a estimativa é positiva (",
      round(estimativa, 4),
      "), a média de E é maior que a média dos grupos A-D.\n"
    )
    
  } else {
    
    cat(
      "Como a estimativa é negativa (",
      round(estimativa, 4),
      "), a média de E é menor que a média dos grupos A-D.\n"
    )
  }
  
  
  if (p_contraste < 0.05) {
    
    cat(
      "O p-valor é",
      signif(p_contraste, 4),
      "e é menor que 0,05.\n"
    )
    
    cat(
      "Portanto, há evidência estatística de diferença entre E\n",
      "e a média dos grupos A-D.\n"
    )
    
  } else {
    
    cat(
      "O p-valor é",
      signif(p_contraste, 4),
      "e é maior ou igual a 0,05.\n"
    )
    
    cat(
      "Portanto, não há evidência estatística suficiente para\n",
      "afirmar que E seja diferente da média dos grupos A-D.\n"
    )
  }
  
  
  # =======================================================
  # TAMANHO DE EFEITO DO CONTRASTE
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("9. TAMANHO DE EFEITO DO CONTRASTE\n")
  cat("------------------------------------------\n")
  
  sigma_modelo <- sigma(modelo)
  
  d_contraste <- estimativa / sigma_modelo
  
  cat(
    "d =",
    round(d_contraste, 4),
    "\n"
  )
  
  
  cat("\nINTERPRETAÇÃO:\n")
  
  cat(
    "O tamanho de efeito d indica a magnitude da diferença\n",
    "entre E e a média dos grupos A-D em unidades padronizadas.\n"
  )
  
  cat(
    "O sinal indica a direção da diferença:\n",
    "positivo = E maior que A-D;\n",
    "negativo = E menor que A-D.\n"
  )
  
  cat(
    "Para avaliar a magnitude, deve-se observar o valor absoluto\n",
    "de d e interpretá-lo juntamente com o intervalo de confiança\n",
    "e o p-valor.\n"
  )
  
  
  # =======================================================
  # CONCLUSÃO AUTOMÁTICA
  # =======================================================
  
  cat("\n")
  cat("------------------------------------------\n")
  cat("10. RESUMO DA ANÁLISE\n")
  cat("------------------------------------------\n")
  
  
  cat("\nANOVA:\n")
  
  if (p_anova < 0.05) {
    cat(
      "→ Há evidência de diferenças entre as condições de umidade.\n"
    )
  } else {
    cat(
      "→ Não há evidência suficiente de diferenças entre as condições.\n"
    )
  }
  
  
  cat("\nNormalidade:\n")
  
  if (shapiro$p.value < 0.05) {
    cat(
      "→ O pressuposto de normalidade dos resíduos NÃO foi atendido.\n"
    )
  } else {
    cat(
      "→ Não foi identificada violação da normalidade dos resíduos.\n"
    )
  }
  
  
  cat("\nHomogeneidade:\n")
  
  if (p_levene < 0.05) {
    cat(
      "→ O pressuposto de homogeneidade das variâncias NÃO foi atendido.\n"
    )
  } else {
    cat(
      "→ Não foi identificada violação da homogeneidade das variâncias.\n"
    )
  }
  
  
  cat("\nContraste E vs A-D:\n")
  
  if (p_contraste < 0.05) {
    cat(
      "→ E apresentou diferença significativa em relação à média de A-D.\n"
    )
  } else {
    cat(
      "→ Não houve evidência suficiente de diferença entre E e a média de A-D.\n"
    )
  }
  
  
  # =======================================================
  # RETORNAR RESULTADOS
  # =======================================================
  
  return(
    list(
      medias = medias,
      modelo = modelo,
      anova = tabela_anova,
      shapiro = shapiro,
      levene = levene,
      eta2 = eta2,
      tukey = tukey,
      contraste_E = resultado_contraste,
      d_contraste = d_contraste
    )
  )
}
