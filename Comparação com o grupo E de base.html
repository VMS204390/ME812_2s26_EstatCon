# ANÁLISE ESTATÍSTICA - MÓDULOS DE ELASTICIDADE

# =========================================================
# 3. FUNÇÃO DE ANÁLISE
# =========================================================

analisar_variavel <- function(dados, variavel) {
  
  
  cat("\n\n")
  cat("====================================================\n")
  cat("ANÁLISE:", variavel, "\n")
  cat("====================================================\n")
  
  
  # -------------------------------------------------------
  # 3.1 ESTATÍSTICA DESCRITIVA + IC95%
  # -------------------------------------------------------
  
  desc <- dados %>%
    group_by(Grupo) %>%
    summarise(
      n = sum(!is.na(.data[[variavel]])),
      media = mean(.data[[variavel]], na.rm = TRUE),
      mediana = median(.data[[variavel]], na.rm = TRUE),
      dp = sd(.data[[variavel]], na.rm = TRUE),
      minimo = min(.data[[variavel]], na.rm = TRUE),
      maximo = max(.data[[variavel]], na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      erro_padrao = dp / sqrt(n),
      t_critico = qt(0.975, df = n - 1),
      IC_inf = media - t_critico * erro_padrao,
      IC_sup = media + t_critico * erro_padrao
    )
  
  
  cat("\n--- Estatística descritiva ---\n")
  print(desc)
  
  
  # -------------------------------------------------------
  # 3.2 MODELO ANOVA
  # -------------------------------------------------------
  
  formula <- as.formula(
    paste(variavel, "~ Grupo")
  )
  
  modelo <- aov(formula, data = dados)
  
  
  # -------------------------------------------------------
  # 3.3 NORMALIDADE DOS RESÍDUOS
  # -------------------------------------------------------
  
  shapiro <- shapiro.test(residuals(modelo))
  p_shapiro <- shapiro$p.value
  
  cat("\n--- Pressuposto de normalidade dos resíduos ---\n")
  cat(
    "Shapiro-Wilk: p =",
    format.pval(p_shapiro),
    "\n"
  )
  
  if (p_shapiro < 0.05) {
    
    cat(
      "Interpretação: Os dados diferem significativamente de uma distribuição normal. Rejeita-se a normalidade.\n",
      sep = ""
    )
    
  } else {
    
    cat(
      "Interpretação: Os dados podem ser considerados normais\n"
    )
  }
  
  
  # -------------------------------------------------------
  # 3.4 HOMOGENEIDADE DAS VARIÂNCIAS
  # -------------------------------------------------------
  
  levene <- leveneTest(formula, data = dados)
  p_levene <- levene$`Pr(>F)`[1]
  
  cat("\n--- Pressuposto de homogeneidade das variâncias ---\n")
  cat(
    "Levene: p =",
    format.pval(p_levene),
    "\n"
  )
  
  if (p_levene < 0.05) {
    
    cat(
      "Interpretação: há evidências de que as variâncias ",
      "dos grupos são diferentes.\n",
      sep = ""
    )
    
  } else {
    
    cat(
      "Interpretação: não há evidências suficientes de ",
      "diferença entre as variâncias.\n"
    )
  }
  
  
  # -------------------------------------------------------
  # 3.5 ANOVA
  # -------------------------------------------------------
  
  anova_resultado <- summary(modelo)
  
  p_anova <- anova_resultado[[1]]$`Pr(>F)`[1]
  
  cat("\n--- ANOVA ---\n")
  print(anova_resultado)
  
  if (p_anova < 0.05) {
    
    cat(
      "\nInterpretação: há evidências de diferença ",
      "entre as médias dos grupos.\n",
      sep = ""
    )
    
  } else {
    
    cat(
      "\nInterpretação: não há evidências suficientes ",
      "de diferença entre as médias dos grupos.\n"
    )
  }
  
  
  # -------------------------------------------------------
  # 3.6 TAMANHO DE EFEITO - ETA²
  # -------------------------------------------------------
  
  tabela_anova <- anova_resultado[[1]]
  
  SQ_grupo <- tabela_anova["Grupo", "Sum Sq"]
  SQ_total <- sum(tabela_anova[, "Sum Sq"])
  
  eta2 <- SQ_grupo / SQ_total
  
  cat("\n--- Tamanho de efeito ---\n")
  cat("Eta² =", round(eta2, 4), "\n")
  
  
  # -------------------------------------------------------
  # 3.7 TUKEY
  # -------------------------------------------------------
  
  tukey <- NULL
  
  if (p_anova < 0.05) {
    
    cat("\n--- Comparações múltiplas de Tukey ---\n")
    
    tukey <- TukeyHSD(modelo)
    
    print(tukey)
    
    tukey_df <- as.data.frame(tukey$Grupo)
    tukey_df$comparacao <- rownames(tukey_df)
    
    significativos <- tukey_df %>%
      filter(`p adj` < 0.05)
    
    cat("\nComparações significativas:\n")
    
    if (nrow(significativos) == 0) {
      
      cat("Nenhum par apresentou diferença significativa.\n")
      
    } else {
      
      for (i in 1:nrow(significativos)) {
        
        cat(
          significativos$comparacao[i],
          "→ p =",
          format.pval(significativos$`p adj`[i]),
          "\n"
        )
      }
    }
    
  } else {
    
    cat(
      "\nTukey não realizado, pois a ANOVA ",
      "não foi significativa.\n",
      sep = ""
    )
  }
  
  
  # -------------------------------------------------------
  # 3.8 CONTRASTE PLANEJADO
  #     E VS MÉDIA DOS GRUPOS A, B, C E D
  # -------------------------------------------------------
  
  cat(
    "\n--- Contraste planejado: E vs. média de A-D ---\n"
  )
  
  medias <- emmeans(modelo, ~ Grupo)
  
  contraste_E <- contrast(
    medias,
    method = list(
      "E vs. A-D" = c(
        -0.25,
        -0.25,
        -0.25,
        -0.25,
        1
      )
    )
  )
  
  contraste_resultado <- summary(
    contraste_E,
    infer = c(TRUE, TRUE)
  )
  
  print(contraste_resultado)
  
  
  # Valores do contraste
  
  estimativa <- contraste_resultado$estimate
  p_contraste <- contraste_resultado$p.value
  IC_inf_contraste <- contraste_resultado$lower.CL
  IC_sup_contraste <- contraste_resultado$upper.CL
  
  
  # -------------------------------------------------------
  # 3.9 INTERPRETAÇÃO DO CONTRASTE
  # -------------------------------------------------------
  
  if (p_contraste < 0.05) {
    
    if (estimativa > 0) {
      
      cat(
        "\nInterpretação: o grupo E apresenta média ",
        "significativamente maior que a média dos grupos ",
        "A, B, C e D.\n",
        sep = ""
      )
      
    } else {
      
      cat(
        "\nInterpretação: o grupo E apresenta média ",
        "significativamente menor que a média dos grupos ",
        "A, B, C e D.\n",
        sep = ""
      )
    }
    
  } else {
    
    cat(
      "\nInterpretação: não há evidências suficientes de ",
      "diferença entre a média do grupo E e a média dos ",
      "grupos A, B, C e D.\n",
      sep = ""
    )
  }
  
  
  # -------------------------------------------------------
  # 3.10 TAMANHO DE EFEITO DO CONTRASTE
  # -------------------------------------------------------
  
  sigma_modelo <- sigma(modelo)
  
  d_contraste <- estimativa / sigma_modelo
  
  cat("\n--- Tamanho de efeito do contraste E vs. A-D ---\n")
  cat(
    "d =",
    round(d_contraste, 4),
    "\n"
  )
  
  
  # -------------------------------------------------------
  # 3.11 OBSERVAÇÃO SOBRE OS PRESSUPOSTOS
  # -------------------------------------------------------
  
  if (p_shapiro < 0.05 | p_levene < 0.05) {
    
    cat(
      "\n--- Observação ---\n"
    )
    
    cat(
      "Um ou mais pressupostos da ANOVA apresentaram ",
      "evidências de violação. Os resultados da ANOVA ",
      "devem, portanto, ser interpretados com cautela.\n",
      sep = ""
    )
  }
  
  
  # -------------------------------------------------------
  # 3.12 GRÁFICOS DE DIAGNÓSTICO
  # -------------------------------------------------------
  
  par(mfrow = c(2, 2))
  plot(modelo)
  par(mfrow = c(1, 1))
  
  
  # -------------------------------------------------------
  # 3.13 RETORNO DOS RESULTADOS
  # -------------------------------------------------------
  
  return(
    list(
      descritiva = desc,
      modelo = modelo,
      shapiro = shapiro,
      levene = levene,
      anova = anova_resultado,
      tukey = tukey,
      contraste_E = contraste_resultado,
      eta2 = eta2,
      d_contraste = d_contraste
    )
  )
}

grafico_media_ic <- function(dados, variavel) {
  
  resumo <- dados %>%
    group_by(Grupo) %>%
    summarise(
      n = sum(!is.na(.data[[variavel]])),
      media = mean(.data[[variavel]], na.rm = TRUE),
      dp = sd(.data[[variavel]], na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      erro = qt(0.975, df = n - 1) * dp / sqrt(n),
      IC_inf = media - erro,
      IC_sup = media + erro
    )
  
  
  ggplot(
    resumo,
    aes(x = Grupo, y = media)
  ) +
    geom_point(size = 3) +
    geom_errorbar(
      aes(
        ymin = IC_inf,
        ymax = IC_sup
      ),
      width = 0.15
    ) +
    labs(
      title = paste(
        "Média de", variavel,
        "por classe de umidade"
      ),
      x = "Classe de umidade",
